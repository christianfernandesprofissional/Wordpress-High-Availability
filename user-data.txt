#!/bin/bash

sudo yum update -y
# Install Docker
sudo yum install docker -y
sudo service docker start
sudo systemctl enable docker
sudo usermod -a -G docker ec2-user
sudo chmod 666 /var/run/docker.sock

# Instala Docker Compose
sudo curl -L https://github.com/docker/compose/releases/download/1.22.0/docker-compose-$(uname -s)-$(uname -m) -o /usr/local/bin/docker-compose
sudo chmod +x /usr/local/bin/docker-compose

mkdir /my-compose

# Adiciona o repositório do MySQL na máquina e instala o MySQL
sudo wget https://dev.mysql.com/get/mysql80-community-release-el9-5.noarch.rpm
sudo dnf install -y https://dev.mysql.com/get/mysql80-community-release-el9-5.noarch.rpm
sudo dnf install -y mysql-community-server

# Configura algumas variáveis
RDS_HOST="Link-do-seu-RDS"
RDS_ADMIN_USER="seu usuário"
RDS_ADMIN_PASSWORD="sua senha"
WP_DB_NAME="wordpress"
WP_USER="wordpress"
WP_PASSWORD="sua senha"

# Entra no banco de dados RDS e cria a tabela wordpress se necessário
mysql -h $RDS_HOST -u $RDS_ADMIN_USER -p$RDS_ADMIN_PASSWORD <<EOF
CREATE DATABASE IF NOT EXISTS $WP_DB_NAME CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
CREATE USER IF NOT EXISTS '$WP_USER'@'%' IDENTIFIED BY '$WP_PASSWORD';
GRANT ALL PRIVILEGES ON $WP_DB_NAME.* TO '$WP_USER'@'%';
FLUSH PRIVILEGES;
EOF

echo "version: '3.3'" >> /my-compose/docker-compose.yml
echo >> /my-compose/docker-compose.yml
echo "services: " >> /my-compose/docker-compose.yml
echo "  wordpress:" >> /my-compose/docker-compose.yml
echo "    image: wordpress:latest" >> /my-compose/docker-compose.yml
echo "    restart: always" >> /my-compose/docker-compose.yml
echo "    ports:" >> /my-compose/docker-compose.yml
echo "      - \"80:80\"" >> /my-compose/docker-compose.yml
echo "    environment:" >> /my-compose/docker-compose.yml
echo "      WORDPRESS_DB_HOST: link-RDS:3306" >> /my-compose/docker-compose.yml
echo "      WORDPRESS_DB_USER: seu usuario" >> /my-compose/docker-compose.yml
echo "      WORDPRESS_DB_PASSWORD: sua senha" >> /my-compose/docker-compose.yml
echo "      WORDPRESS_DB_NAME: nome do seu banco de dados" >> /my-compose/docker-compose.yml
echo "    volumes:" >> /my-compose/docker-compose.yml
echo "      - wordpress:/var/www/html" >> /my-compose/docker-compose.yml
echo >> /my-compose/docker-compose.yml
echo "volumes:" >> /my-compose/docker-compose.yml
echo "  wordpress:" >> /my-compose/docker-compose.yml

cd /my-compose
docker-compose up -d
